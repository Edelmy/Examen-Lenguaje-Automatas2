def factorial(n):
    if n==0 or n==1:
        resultado=1
    elif n>1:
        resultado=n*factorial(n-1)
    return resultado

fact5=factorial(5)
print(fact5)

a) Polaca Inversa (de cada expresión):
   n-1  ->  n 1 -
   n*T2  ->  n T2 *
   n==0  ->  n 0 ==
   n==1  ->  n 1 ==
   n>1  ->  n 1 >

b) Tres Direcciones:
   func factorial(n)
       if n == 0 goto L1
       if n == 1 goto L1
       if n > 1 goto L2
       goto L3
   L1:
       resultado = 1
       goto L4
   L2:
       T1 = n - 1
       param T1
       T2 = call factorial, 1
       T3 = n * T2
       resultado = T3
       goto L4
   L3:
   L4:
       return resultado
   endfunc
   param 5
   T4 = call factorial, 1
   fact5 = T4
   param fact5
   call print, 1

c) Cuadrupletas:
   (func, factorial, n, -)
   (if==, n, 0, L1)
   (if==, n, 1, L1)
   (if>, n, 1, L2)
   (goto, -, -, L3)
   (label, -, -, L1)
   (=, 1, -, resultado)
   (goto, -, -, L4)
   (label, -, -, L2)
   (-, n, 1, T1)
   (param, T1, -, -)
   (call, factorial, 1, T2)
   (*, n, T2, T3)
   (=, T3, -, resultado)
   (goto, -, -, L4)
   (label, -, -, L3)
   (label, -, -, L4)
   (return, resultado, -, -)
   (endfunc, -, -, -)
   (param, 5, -, -)
